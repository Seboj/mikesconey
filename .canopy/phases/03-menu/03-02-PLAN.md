Add a `<script>` tag at the bottom of menu.astro (after the BaseLayout closing tag is NOT possible in Astro — place it inside the BaseLayout slot, after the menu content div) that implements:

**1. Scroll-spy with IntersectionObserver:**

```javascript
document.addEventListener('DOMContentLoaded', () => {
  const sections = document.querySelectorAll('[data-menu-section]');
  const navLinks = document.querySelectorAll('[data-category-link]');

  if (sections.length === 0 || navLinks.length === 0) return;

  const observer = new IntersectionObserver(
    (entries) => {
      entries.forEach((entry) => {
        if (entry.isIntersecting) {
          const id = entry.target.id;
          navLinks.forEach((link) => {
            const isActive = link.getAttribute('href') === `#${id}`;
            link.classList.toggle('text-brand-primary', isActive);
            link.classList.toggle('bg-brand-accent', isActive);
            link.classList.toggle('text-brand-text-muted', !isActive);
          });
        }
      });
    },
    {
      // Negative top margin accounts for sticky headers (site nav ~68px + category nav ~48px ≈ 116px)
      // -70% bottom means a section is "active" when it occupies the upper 30% of viewport
      rootMargin: '-120px 0px -70% 0px',
      threshold: 0
    }
  );

  sections.forEach((section) => observer.observe(section));
```

**2. Smooth scroll on category link click:**

```javascript
  navLinks.forEach((link) => {
    link.addEventListener('click', (e) => {
      e.preventDefault();
      const targetId = link.getAttribute('href')?.slice(1);
      const target = targetId ? document.getElementById(targetId) : null;
      if (target) {
        target.scrollIntoView({ behavior: 'smooth', block: 'start' });
        // Update URL hash without jumping
        history.pushState(null, '', `#${targetId}`);
      }
    });
  });
```

**3. Scroll active link into view in the category nav:**

When scroll-spy activates a new category, if that category link is not fully visible in the horizontal scrollable nav, scroll the nav container to make it visible:

```javascript
  // Auto-scroll the category nav to show the active link
  function scrollNavToActive() {
    const activeLink = document.querySelector('[data-category-link].text-brand-primary');
    if (activeLink) {
      const nav = activeLink.closest('.overflow-x-auto');
      if (nav) {
        const linkLeft = activeLink.offsetLeft;
        const linkWidth = activeLink.offsetWidth;
        const navScroll = nav.scrollLeft;
        const navWidth = nav.offsetWidth;

        if (linkLeft < navScroll || linkLeft + linkWidth > navScroll + navWidth) {
          nav.scrollTo({
            left: linkLeft - navWidth / 2 + linkWidth / 2,
            behavior: 'smooth'
          });
        }
      }
    }
  }
```

Call `scrollNavToActive()` inside the IntersectionObserver callback after updating active classes.

Close the DOMContentLoaded listener.

**4. Respect prefers-reduced-motion:**

At the top of the script, check for reduced motion preference:
```javascript
const prefersReducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
```
Use `behavior: prefersReducedMotion ? 'auto' : 'smooth'` in both scrollIntoView and scrollTo calls.

IMPORTANT: Place the `<script>` tag inside the BaseLayout slot but after the menu content div. In Astro, scripts in .astro files are bundled and work correctly within the page.
